$ ===== MULTI-HEAD ATTENTION =====

$ Cabeza 1: Verbos
logic def cabeza_verbo (1 = es_verbo)
    var query_verbo, 0.0
    var key_verbo, 0.0
    var value_verbo, 0.0
    var attn_verbo, 0.0
    
    proc query_verbo, embed
    proc key_verbo, embed
    proc value_verbo, embed
    conv attn_verbo, query_verbo, key_verbo, 0.0
    proc value_verbo, attn_verbo
logic endef

var ff1, 0.0

proc ff2, ff1
